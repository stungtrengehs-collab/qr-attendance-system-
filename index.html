កូដ Google Apps Script ទម្រង់ "Set-and-Forget" (អាន Config ពី Google Sheet)

កូដនេះនឹងអានទិន្នន័យ Bot Token & Chat ID ចេញពី Sheet `BotConfig` ដោយស្វ័យប្រវត្តិ។ អ្នកមិនបាច់ Deploy ឡើងវិញទេពេលផ្លាស់ប្តូរ Bot តាមថ្នាក់!

/**
 * Google Apps Script - Set-and-Forget Attendance System
 * ----------------------------------------------------
 * ១. Sheet "Attendance": សម្រាប់កត់ត្រាវត្តមានសិស្ស
 * ២. Sheet "BotConfig": សម្រាប់កំណត់ Bot Token & Group Chat ID តាមថ្នាក់
 *    - Col A: ClassName (ឧ. ថ្នាក់ទី១២A)
 *    - Col B: BotToken  (ឧ. 123456789:ABC...)
 *    - Col C: ChatID    (ឧ. -100123456789)
 */

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var attendanceSheet = ss.getSheetByName("Attendance") || ss.getActiveSheet();
    
    var data = JSON.parse(e.postData.contents);

    var studentId   = data.student_id;
    var studentName = data.student_name;
    var className   = data.class_name;
    var subject     = data.subject;
    var teacher     = data.teacher || "គ្រូបង្រៀន";
    var status      = data.status || "មកដល់";
    var timeStr     = data.timestamp || new Date().toLocaleTimeString();

    // 1. រក្សាទុកវត្តមានចូល Sheet "Attendance"
    attendanceSheet.appendRow([new Date(), studentId, studentName, className, subject, teacher, status]);

    // 2. ស្វែងរក Bot Token និង Chat ID ពីក្នុង Sheet "BotConfig" ឬ Payload
    var botConfig = getBotConfigForClass(ss, className);
    var botToken  = data.telegram_token || botConfig.token;
    var chatId    = data.telegram_chat_id || botConfig.chat_id;

    // 3. ប្រសិនបើមាន Bot Config នោះនឹងផ្ញើសារទៅ Telegram
    if (botToken && chatId) {
      sendClassTelegramNotification(botToken, chatId, studentId, studentName, className, subject, teacher, status, timeStr);
    }

    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
                         .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}

// មុខងារអាន Bot Token & Chat ID ចេញពី Sheet "BotConfig" ដោយស្វ័យប្រវត្តិ
function getBotConfigForClass(ss, className) {
  var configSheet = ss.getSheetByName("BotConfig");
  var result = { token: "", chat_id: "" };
  
  if (!configSheet) return result;

  var data = configSheet.getDataRange().getValues();
  // រំលង Header ជួរដេកទី ១
  for (var i = 1; i < data.length; i++) {
    var sheetClass = data[i][0] ? data[i][0].toString().trim() : "";
    if (sheetClass === className.trim()) {
      result.token   = data[i][1] ? data[i][1].toString().trim() : "";
      result.chat_id = data[i][2] ? data[i][2].toString().trim() : "";
      break;
    }
  }
  return result;
}

function sendClassTelegramNotification(token, chatId, id, name, cls, subj, teacher, status, timeStr) {
  var url = "https://api.telegram.org/bot" + token + "/sendMessage";
  
  var text = "🚨 **វត្តមានសិស្សតាមថ្នាក់រៀន**\n\n" +
             "🏫 **ថ្នាក់រៀន៖** " + cls + "\n" +
             "📚 **មុខវិជ្ជា៖** " + subj + "\n" +
             "👨‍🏫 **គ្រូបង្រៀន៖** " + teacher + "\n\n" +
             "🆔 **អត្តលេខ៖** " + id + "\n" +
             "👤 **ឈ្មោះសិស្ស៖** " + name + "\n" +
             "⏱ **ម៉ោងស្កេន៖** " + timeStr + "\n" +
             "✅ **ស្ថានភាព៖** " + status;

  var payload = {
    "chat_id": chatId,
    "text": text,
    "parse_mode": "Markdown"
  };

  var options = {
    "method": "post",
    "contentType": "application/json",
    "payload": JSON.stringify(payload)
  };

  UrlFetchApp.fetch(url, options);
}
